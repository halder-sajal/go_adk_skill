package sessiontestsuite // import "google.golang.org/adk/v2/session/sessiontestsuite"

Package sessiontestsuite provides a reusable, backend-agnostic test suite for
verifying session.Service implementations.

FUNCTIONS

func RunServiceTests(t *testing.T, opts SuiteOptions, setup func(t *testing.T) session.Service)
    RunServiceTests runs a battery of standard tests against a Session.Service.
